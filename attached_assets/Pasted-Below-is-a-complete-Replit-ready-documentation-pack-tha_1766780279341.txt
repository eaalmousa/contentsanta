4) Database Schema (Postgres SQL)

Create one migration file first: infra/migrations/001_init.sql

This schema is multi-tenant, includes versioning, approvals, runs, and usage ledger.

-- 001_init.sql

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ENUMS
DO $$ BEGIN
  CREATE TYPE role_type AS ENUM ('owner','admin','editor','reviewer','viewer');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE input_type AS ENUM ('url','text','pdf','docx','transcript','brief');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE asset_status AS ENUM ('draft','in_review','approved','published','archived');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE run_status AS ENUM ('queued','running','succeeded','failed','cancelled');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE channel_type AS ENUM ('blog','press_release','linkedin','x','instagram','newsletter','website','generic');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE target_type AS ENUM ('wordpress','webflow','x','linkedin','meta','email','custom');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE publish_status AS ENUM ('queued','scheduled','running','succeeded','failed','cancelled');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE usage_unit_type AS ENUM ('run','tokens','seconds');
EXCEPTION WHEN duplicate_object THEN null;
END $$;

-- CORE
CREATE TABLE IF NOT EXISTS users (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  email text UNIQUE NOT NULL,
  full_name text,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS workspaces (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  name text NOT NULL,
  slug text UNIQUE NOT NULL,
  plan_id text,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS workspace_users (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  user_id uuid NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  role role_type NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE(workspace_id, user_id)
);

CREATE TABLE IF NOT EXISTS brands (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  name text NOT NULL,
  default_language text NOT NULL DEFAULT 'en',
  brand_rules_json jsonb NOT NULL DEFAULT '{}'::jsonb,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE(workspace_id, name)
);

CREATE TABLE IF NOT EXISTS templates (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid REFERENCES brands(id) ON DELETE CASCADE,
  type text NOT NULL,             -- e.g. "press_release", "seo_blog", "social_pack"
  name text NOT NULL,
  content text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS projects (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid NOT NULL REFERENCES brands(id) ON DELETE CASCADE,
  name text NOT NULL,
  description text,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS inputs (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid NOT NULL REFERENCES brands(id) ON DELETE CASCADE,
  project_id uuid REFERENCES projects(id) ON DELETE SET NULL,
  type input_type NOT NULL,
  title text,
  source_url text,
  language text,
  raw_text text,
  file_key text,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- AI WORKFLOWS
CREATE TABLE IF NOT EXISTS workflows (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid REFERENCES brands(id) ON DELETE CASCADE,
  name text NOT NULL,
  type text NOT NULL,             -- e.g. seo_blog_v1, press_release_v1
  config_json jsonb NOT NULL DEFAULT '{}'::jsonb,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS workflow_runs (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid NOT NULL REFERENCES brands(id) ON DELETE CASCADE,
  input_id uuid NOT NULL REFERENCES inputs(id) ON DELETE CASCADE,
  workflow_type text NOT NULL,
  status run_status NOT NULL DEFAULT 'queued',
  started_at timestamptz,
  completed_at timestamptz,
  cost_estimate numeric,
  trace_json jsonb NOT NULL DEFAULT '{}'::jsonb,
  output_json jsonb NOT NULL DEFAULT '{}'::jsonb,
  error_json jsonb,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- ASSETS + VERSIONING
CREATE TABLE IF NOT EXISTS assets (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid NOT NULL REFERENCES brands(id) ON DELETE CASCADE,
  project_id uuid REFERENCES projects(id) ON DELETE SET NULL,
  input_id uuid REFERENCES inputs(id) ON DELETE SET NULL,
  status asset_status NOT NULL DEFAULT 'draft',
  primary_language text NOT NULL DEFAULT 'en',
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS asset_versions (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  asset_id uuid NOT NULL REFERENCES assets(id) ON DELETE CASCADE,
  version_no int NOT NULL,
  title text,
  body text NOT NULL,
  format text NOT NULL DEFAULT 'md',         -- md/html/plain
  channel channel_type NOT NULL DEFAULT 'generic',
  language text NOT NULL DEFAULT 'en',
  metadata_json jsonb NOT NULL DEFAULT '{}'::jsonb,
  created_by uuid REFERENCES users(id) ON DELETE SET NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE(asset_id, version_no)
);

CREATE TABLE IF NOT EXISTS comments (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  asset_version_id uuid NOT NULL REFERENCES asset_versions(id) ON DELETE CASCADE,
  user_id uuid NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  body text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- PUBLISHING
CREATE TABLE IF NOT EXISTS publishing_targets (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  brand_id uuid NOT NULL REFERENCES brands(id) ON DELETE CASCADE,
  type target_type NOT NULL,
  name text NOT NULL,
  credentials_encrypted text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS publish_jobs (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  asset_version_id uuid NOT NULL REFERENCES asset_versions(id) ON DELETE CASCADE,
  target_id uuid NOT NULL REFERENCES publishing_targets(id) ON DELETE CASCADE,
  status publish_status NOT NULL DEFAULT 'queued',
  scheduled_at timestamptz,
  executed_at timestamptz,
  result_json jsonb,
  error_json jsonb,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- USAGE
CREATE TABLE IF NOT EXISTS usage_ledger (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id uuid NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  user_id uuid REFERENCES users(id) ON DELETE SET NULL,
  run_id uuid REFERENCES workflow_runs(id) ON DELETE SET NULL,
  units numeric NOT NULL,
  unit_type usage_unit_type NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- INDEXES (performance)
CREATE INDEX IF NOT EXISTS idx_workspace_users_workspace ON workspace_users(workspace_id);
CREATE INDEX IF NOT EXISTS idx_brands_workspace ON brands(workspace_id);
CREATE INDEX IF NOT EXISTS idx_inputs_workspace ON inputs(workspace_id);
CREATE INDEX IF NOT EXISTS idx_runs_workspace ON workflow_runs(workspace_id);
CREATE INDEX IF NOT EXISTS idx_assets_workspace ON assets(workspace_id);
CREATE INDEX IF NOT EXISTS idx_versions_asset ON asset_versions(asset_id);

{ workspaceId, brandId, inputId, workflowType, options }

GET /assets?workspaceId=&brandId=&status=&q=

POST /export/:assetVersionId → returns downloadable file (MD/HTML/DOCX/PDF)

6) AI Workflow Runner Design (Agent Builder)
6.1 Standard Payload into Agent

Backend must assemble:

{
  "workflowType": "press_release_v1",
  "input": { "title": "...", "rawText": "...", "sourceUrl": "...", "language": "en" },
  "brand": { "name": "...", "defaultLanguage": "en", "rules": { ... } },
  "options": { "channel": "press_release", "tone": "institutional", "length": "standard" }
}

6.2 Standard Output (must be structured)

Agent returns:

{
  "title": "...",
  "body": "...",
  "metadata": {
    "language": "en",
    "channel": "press_release",
    "keyMessages": [],
    "ctaVariants": [],
    "seo": { "metaTitle": "", "metaDescription": "", "keywords": [] }
  },
  "qualityFlags": {
    "claimsToVerify": [],
    "sensitiveTopics": [],
    "complianceWarnings": []
  },
  "trace": {
    "promptVersion": "press_release_v1.0",
    "model": "...",
    "steps": []
  }
}


Store this in workflow_runs.output_json + trace_json, then convert to asset_versions.

7) Security & Tenant Isolation Rules

Backend must enforce:

every query filtered by workspace_id

user must belong to workspace in workspace_users

role-gated actions:

create brands/templates/projects: admin+

approve/publish: reviewer/admin/owner

admin console: owner only

publishing credentials encrypted (AES-GCM) using ENCRYPTION_KEY
